program calculadora;
var x, y, resultado : integer;
var continuar : boolean;
begin
    read(x, y);
    resultado := x + y;
    write("Soma: ", resultado);
    writeln;
    if resultado > 100 then
        begin
            write("Resultado muito grande!");
            writeln;
        end
    else
        begin
            write("Resultado normal");
            writeln;
        end;
    continuar := true;
    while continuar do
        begin
            readln(x);
            if x = 0 then
                continuar := false
            else
                begin
                    resultado := resultado + x;
                    write("Novo resultado: ", resultado);
                    writeln;
                end;
        end;
end.
